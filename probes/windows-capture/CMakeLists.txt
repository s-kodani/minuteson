cmake_minimum_required(VERSION 3.21)
project(mon_windows_capture LANGUAGES CXX)

include(CTest)
if(BUILD_TESTING)
  add_executable(mon-windows-metadata-tests metadata_worker_tests.cpp)
  target_compile_features(mon-windows-metadata-tests PRIVATE cxx_std_20)
  target_include_directories(mon-windows-metadata-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../.cache/windows-capture-headers")
  add_test(NAME metadata-worker COMMAND mon-windows-metadata-tests)
  add_executable(mon-windows-clock-tests packet_clock_tests.cpp packet_clock.h)
  target_compile_features(mon-windows-clock-tests PRIVATE cxx_std_20)
  add_test(NAME packet-clock COMMAND mon-windows-clock-tests)
  add_executable(mon-windows-queue-tests packet_queue_tests.cpp packet_queue.h)
  target_compile_features(mon-windows-queue-tests PRIVATE cxx_std_20)
  add_test(NAME packet-queue COMMAND mon-windows-queue-tests)
  add_executable(mon-windows-async-tests async_capture_tests.cpp)
  target_compile_features(mon-windows-async-tests PRIVATE cxx_std_20)
  target_include_directories(mon-windows-async-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../.cache/windows-capture-headers")
  target_compile_definitions(mon-windows-async-tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(mon-windows-async-tests PRIVATE ole32 uuid mmdevapi propsys mfplat mfreadwrite mfuuid bcrypt avrt)
  add_test(NAME async-capture COMMAND mon-windows-async-tests)
  if(MSVC)
    target_compile_options(mon-windows-queue-tests PRIVATE /utf-8)
    target_compile_options(mon-windows-async-tests PRIVATE /utf-8)
  endif()
endif()

add_executable(mon-windows-capture main.cpp segmented_wave_writer.h)
target_include_directories(mon-windows-capture PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../.cache/windows-capture-headers")
target_compile_features(mon-windows-capture PRIVATE cxx_std_20)
target_compile_definitions(mon-windows-capture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(mon-windows-capture PRIVATE ole32 uuid mmdevapi propsys mfplat mfreadwrite mfuuid bcrypt avrt)

if(MSVC)
  # The QA installer must run on PCs without the Visual Studio development CRT.
  set_property(TARGET mon-windows-capture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(mon-windows-capture PRIVATE /utf-8 /W4 /permissive-)
endif()
