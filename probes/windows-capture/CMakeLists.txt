cmake_minimum_required(VERSION 3.21)
project(mon_windows_capture LANGUAGES CXX)
if(MSVC)
  # Shared Rust staticlib uses the static CRT; native contract tests must agree.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

find_program(MON_CARGO cargo REQUIRED)
get_filename_component(MON_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(MON_CAPTURE_RUST_DIR "${CMAKE_CURRENT_BINARY_DIR}/capture-core")
if(WIN32)
  set(MON_CAPTURE_STATIC "${MON_CAPTURE_RUST_DIR}/x86_64-pc-windows-msvc/release/mon_capture_ffi.lib")
  set(MON_CAPTURE_CARGO_PREFIX "${CMAKE_COMMAND}" -E env "CARGO_ENCODED_RUSTFLAGS=-Ctarget-feature=+crt-static")
  set(MON_CAPTURE_RUST_TARGET --target x86_64-pc-windows-msvc)
else()
  set(MON_CAPTURE_STATIC "${MON_CAPTURE_RUST_DIR}/release/libmon_capture_ffi.a")
endif()
add_custom_target(mon-capture-core-build
  COMMAND ${MON_CAPTURE_CARGO_PREFIX} "${MON_CARGO}" build --manifest-path "${MON_ROOT}/Cargo.toml" -p mon-capture-ffi --release --locked --target-dir "${MON_CAPTURE_RUST_DIR}" ${MON_CAPTURE_RUST_TARGET}
  BYPRODUCTS "${MON_CAPTURE_STATIC}"
  COMMENT "Build shared Rust capture contracts")
add_library(mon-capture-core STATIC IMPORTED GLOBAL)
set_target_properties(mon-capture-core PROPERTIES IMPORTED_LOCATION "${MON_CAPTURE_STATIC}")
add_dependencies(mon-capture-core mon-capture-core-build)
if(WIN32)
  set_property(TARGET mon-capture-core PROPERTY INTERFACE_LINK_LIBRARIES bcrypt userenv ws2_32 ntdll)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(mon-capture-synthetic-tests synthetic_driver_tests.cpp)
  target_compile_features(mon-capture-synthetic-tests PRIVATE cxx_std_20)
  add_test(NAME synthetic-driver COMMAND mon-capture-synthetic-tests)
  add_executable(mon-capture-core-tests capture_core_tests.cpp)
  target_compile_features(mon-capture-core-tests PRIVATE cxx_std_20)
  target_link_libraries(mon-capture-core-tests PRIVATE mon-capture-core)
  add_test(NAME capture-core COMMAND mon-capture-core-tests)
  add_executable(mon-m4a-container-tests m4a_container_tests.cpp m4a_container.h)
  target_link_libraries(mon-m4a-container-tests PRIVATE mon-capture-core)
  target_compile_features(mon-m4a-container-tests PRIVATE cxx_std_20)
  add_test(NAME m4a-container COMMAND mon-m4a-container-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures/audio-import")
  add_executable(mon-windows-metadata-tests metadata_worker_tests.cpp)
  target_compile_features(mon-windows-metadata-tests PRIVATE cxx_std_20)
  target_include_directories(mon-windows-metadata-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../.cache/windows-capture-headers")
  add_test(NAME metadata-worker COMMAND mon-windows-metadata-tests)
  add_executable(mon-windows-clock-tests packet_clock_tests.cpp packet_clock.h)
  target_compile_features(mon-windows-clock-tests PRIVATE cxx_std_20)
  add_test(NAME packet-clock COMMAND mon-windows-clock-tests)
  add_executable(mon-windows-queue-tests packet_queue_tests.cpp packet_queue.h)
  target_link_libraries(mon-windows-queue-tests PRIVATE mon-capture-core)
  target_compile_features(mon-windows-queue-tests PRIVATE cxx_std_20)
  add_test(NAME packet-queue COMMAND mon-windows-queue-tests)
  add_executable(mon-windows-preparation-tests input_preparation_tests.cpp input_preparation.h loopback_scope.h)
  target_link_libraries(mon-windows-preparation-tests PRIVATE mon-capture-core)
  target_compile_features(mon-windows-preparation-tests PRIVATE cxx_std_20)
  add_test(NAME input-preparation COMMAND mon-windows-preparation-tests)
  add_executable(mon-windows-async-tests async_capture_tests.cpp)
  target_link_libraries(mon-windows-async-tests PRIVATE mon-capture-core)
  target_compile_features(mon-windows-async-tests PRIVATE cxx_std_20)
  target_include_directories(mon-windows-async-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../.cache/windows-capture-headers")
  target_compile_definitions(mon-windows-async-tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(mon-windows-async-tests PRIVATE ole32 uuid mmdevapi propsys mfplat mfreadwrite mfuuid bcrypt avrt)
  add_test(NAME async-capture COMMAND mon-windows-async-tests)
  if(MSVC)
    target_compile_options(mon-windows-queue-tests PRIVATE /utf-8)
    target_compile_options(mon-windows-async-tests PRIVATE /utf-8)
  endif()
endif()

add_executable(mon-windows-capture main.cpp segmented_wave_writer.h)
target_link_libraries(mon-windows-capture PRIVATE mon-capture-core)
target_include_directories(mon-windows-capture PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../.cache/windows-capture-headers")
target_compile_features(mon-windows-capture PRIVATE cxx_std_20)
target_compile_definitions(mon-windows-capture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(mon-windows-capture PRIVATE ole32 uuid mmdevapi propsys mfplat mfreadwrite mfuuid bcrypt avrt)

if(MSVC)
  # The QA installer must run on PCs without the Visual Studio development CRT.
  set_property(TARGET mon-windows-capture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(mon-windows-capture PRIVATE /utf-8 /W4 /permissive-)
endif()
